function enhanceImage() { if (canvas.width === 0) { alert( "Please scan an ID Card first." ); return; } status.innerText = "Enhancing ID Card..."; const imgData = ctx.getImageData( 0, 0, canvas.width, canvas.height ); const data = imgData.data; /* Professional enhancement settings */ const contrast = 1.18; const brightness = 8; const saturation = 1.05; /* Calculate average brightness */ let total = 0; for ( let i = 0; i < data.length; i += 4 ) { const gray = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2]; total += gray; } const pixels = data.length / 4; const average = total / pixels; /* Automatic brightness correction */ let autoBrightness = 128 - average; autoBrightness = Math.max( -20, Math.min( 20, autoBrightness * 0.35 ) ); /* Apply enhancement */ for ( let i = 0; i < data.length; i += 4 ) { let r = data[i]; let g = data[i + 1]; let b = data[i + 2]; /* Brightness */ r += brightness + autoBrightness; g += brightness + autoBrightness; b += brightness + autoBrightness; /* Contrast */ r = (r - 128) * contrast + 128; g = (g - 128) * contrast + 128; b = (b - 128) * contrast + 128; /* Slight color enhancement */ const gray = 0.299 * r + 0.587 * g + 0.114 * b; r = gray + (r - gray) * saturation; g = gray + (g - gray) * saturation; b = gray + (b - gray) * saturation; /* Keep RGB between 0 and 255 */ data[i] = Math.max( 0, Math.min( 255, r ) ); data[i + 1] = Math.max( 0, Math.min( 255, g ) ); data[i + 2] = Math.max( 0, Math.min( 255, b ) ); } ctx.putImageData( imgData, 0, 0 ); /* Extra sharpening using a simple convolution. */ sharpenCanvas(); status.innerText = "✓ ID Card Enhanced & Sharpened"; } function sharpenCanvas() { const width = canvas.width; const height = canvas.height; const source = ctx.getImageData( 0, 0, width, height ); const output = ctx.createImageData( width, height ); const src = source.data; const dst = output.data; /* Sharpening kernel */ const kernel = [ 0, -1, 0, -1, 5, -1, 0, -1, 0 ]; for ( let y = 1; y < height - 1; y++ ) { for ( let x = 1; x < width - 1; x++ ) { let r = 0; let g = 0; let b = 0; let k = 0; for ( let ky = -1; ky <= 1; ky++ ) { for ( let kx = -1; kx <= 1; kx++ ) { const px = x + kx; const py = y + ky; const index = (py * width + px) * 4; const weight = kernel[k]; r += src[index] * weight; g += src[index + 1] * weight; b += src[index + 2] * weight; k++; } } const index = (y * width + x) * 4; dst[index] = Math.max( 0, Math.min( 255, r ) ); dst[index + 1] = Math.max( 0, Math.min( 255, g ) ); dst[index + 2] = Math.max( 0, Math.min( 255, b ) ); dst[index + 3] = 255; } } ctx.putImageData( output, 0, 0 ); }
