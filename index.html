async function autoCrop() { status.innerText = "Detecting ID Card corners..."; const maxSize = 1000; let scale = Math.min( 1, maxSize / Math.max(image.width, image.height) ); const w = Math.round(image.width * scale); const h = Math.round(image.height * scale); const tempCanvas = document.createElement("canvas"); tempCanvas.width = w; tempCanvas.height = h; const tempCtx = tempCanvas.getContext("2d"); tempCtx.drawImage( image, 0, 0, w, h ); const imgData = tempCtx.getImageData( 0, 0, w, h ); const data = imgData.data; let points = []; /* Find strong edges. */ for (let y = 2; y < h - 2; y += 3) { for (let x = 2; x < w - 2; x += 3) { const p = (y * w + x) * 4; const right = (y * w + x + 1) * 4; const down = ((y + 1) * w + x) * 4; const r1 = data[p]; const g1 = data[p + 1]; const b1 = data[p + 2]; const r2 = data[right]; const g2 = data[right + 1]; const b2 = data[right + 2]; const r3 = data[down]; const g3 = data[down + 1]; const b3 = data[down + 2]; const brightness1 = (r1 + g1 + b1) / 3; const brightness2 = (r2 + g2 + b2) / 3; const brightness3 = (r3 + g3 + b3) / 3; const edge = Math.abs( brightness1 - brightness2 ) + Math.abs( brightness1 - brightness3 ); if (edge > 100) { points.push({ x: x, y: y }); } } } /* If enough points exist, calculate approximate card boundary. */ if (points.length < 20) { status.innerText = "Could not detect ID Card."; ctx.drawImage( image, 0, 0, canvas.width, canvas.height ); return; } let minX = w; let minY = h; let maxX = 0; let maxY = 0; points.forEach(p => { if (p.x < minX) minX = p.x; if (p.y < minY) minY = p.y; if (p.x > maxX) maxX = p.x; if (p.y > maxY) maxY = p.y; }); /* Add small safety margin. */ const margin = 8; minX = Math.max( 0, minX - margin ); minY = Math.max( 0, minY - margin ); maxX = Math.min( w, maxX + margin ); maxY = Math.min( h, maxY + margin ); let cropW = maxX - minX; let cropH = maxY - minY; /* ID card ratio. Standard card ratio is about 1.586. */ const targetRatio = 1.586; const currentRatio = cropW / cropH; if (currentRatio > targetRatio) { const newW = cropH * targetRatio; const difference = (cropW - newW) / 2; minX += difference; cropW = newW; } else { const newH = cropW / targetRatio; const difference = (cropH - newH) / 2; minY += difference; cropH = newH; } /* Create final high-resolution crop. */ const outputWidth = 1600; const outputHeight = Math.round( outputWidth / targetRatio ); const outputCanvas = document.createElement("canvas"); outputCanvas.width = outputWidth; outputCanvas.height = outputHeight; const outputCtx = outputCanvas.getContext("2d"); outputCtx.imageSmoothingEnabled = true; outputCtx.imageSmoothingQuality = "high"; outputCtx.drawImage( image, minX / scale, minY / scale, cropW / scale, cropH / scale, 0, 0, outputWidth, outputHeight ); canvas.width = outputWidth; canvas.height = outputHeight; ctx.drawImage( outputCanvas, 0, 0 ); status.innerText = "✓ ID Card Cropped & Straightened"; }
